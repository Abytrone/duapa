import 'package:duapa/controllers/password_reset_controller.dart';
import 'package:duapa/models/user_class.dart';
import 'package:duapa/services/constants.dart';
import 'package:duapa/services/theme.dart';
import 'package:duapa/views/widgets/empty.dart';
import 'package:duapa/views/widgets/input_field.dart';
import 'package:duapa/views/widgets/loading_widget.dart';
import 'package:duapa/views/widgets/primary_button.dart';
import 'package:flutter/material.dart';
import 'package:get/get.dart';
import 'package:loading_indicator/loading_indicator.dart';

class SecurityQuestionsScreen extends StatefulWidget {
  const SecurityQuestionsScreen({
    Key? key,
    required this.token,
    required this.user,
  }) : super(key: key);
  final String token;
  final User user;
  @override
  State<SecurityQuestionsScreen> createState() =>
      _SecurityQuestionsScreenState();
}

class _SecurityQuestionsScreenState extends State<SecurityQuestionsScreen> {
  @override
  void initState() {
    super.initState();
  }

  final TextEditingController answerOne = TextEditingController();
  final TextEditingController answerTwo = TextEditingController();
  final TextEditingController answerThree = TextEditingController();
  final GlobalKey<FormState>? formKey = GlobalKey<FormState>();

  final PasswordResetController passwordResetController =
      Get.put(PasswordResetController());

  bool isLoading = false;
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text(
          'Add Security Questions',
          style: TextStyle(
            fontSize: 12,
            color: Theme.of(context).brightness ==
                    lightThemeData(context).brightness
                ? kContentColorLightTheme
                : kContentColorDarkTheme,
          ),
        ),
        centerTitle: true,
        backgroundColor:
            Theme.of(context).brightness == lightThemeData(context).brightness
                ? kContentColorDarkTheme
                : kContentColorLightTheme,
        iconTheme: IconThemeData(
          color:
              Theme.of(context).brightness == lightThemeData(context).brightness
                  ? kContentColorLightTheme
                  : kContentColorDarkTheme,
        ),
      ),
      backgroundColor:
          Theme.of(context).brightness == lightThemeData(context).brightness
              ? kContentColorDarkTheme
              : kContentColorLightTheme,
      body: Obx(
        () => passwordResetController.isLoading.isTrue
            ? const AbyLoading()
            : passwordResetController.questions.value.value != null
                ? ListView.builder(
                    itemCount: 1,
                    shrinkWrap: true,
                    itemBuilder: (BuildContext context, int index) {
                      return Form(
                        key: formKey,
                        child: Column(
                          children: [
                            Column(
                              crossAxisAlignment: CrossAxisAlignment.start,
                              children: <Widget>[
                                const SizedBox(
                                  height: 20,
                                ),
                                Container(
                                  padding: const EdgeInsets.only(left: 30),
                                  child: Text(
                                    passwordResetController
                                        .questions.value.value![0].question,
                                    style: TextStyle(
                                      fontSize: 14,
                                      color: Theme.of(context).brightness ==
                                              lightThemeData(context).brightness
                                          ? kContentColorLightTheme
                                          : kContentColorDarkTheme,
                                    ),
                                  ),
                                ),

                                /// TextFromField Old password
                                Padding(
                                  padding:
                                      const EdgeInsets.symmetric(vertical: 5.0),
                                  child: TextFromField(
                                    icon: Icons.question_answer_outlined,
                                    enabled: isLoading ? false : true,
                                    validate: (value) {
                                      if (value == null || value.isEmpty) {
                                        return "* Required";
                                      } else {
                                        return null;
                                      }
                                    },
                                    label: "Answer",
                                    secured: false,
                                    textCapitalization: TextCapitalization.none,
                                    inputType: TextInputType.text,
                                    controller: answerOne,
                                    maxLines: 1,
                                    leftPadding: 20.0,
                                  ),
                                ),
                                Container(
                                  padding:
                                      const EdgeInsets.only(left: 30, top: 20),
                                  child: Text(
                                    passwordResetController
                                        .questions.value.value![1].question,
                                    style: TextStyle(
                                      fontSize: 14,
                                      color: Theme.of(context).brightness ==
                                              lightThemeData(context).brightness
                                          ? kContentColorLightTheme
                                          : kContentColorDarkTheme,
                                    ),
                                  ),
                                ),

                                /// TextFromField New password
                                Padding(
                                  padding:
                                      const EdgeInsets.symmetric(vertical: 5.0),
                                  child: TextFromField(
                                    icon: Icons.question_answer_outlined,
                                    enabled: isLoading ? false : true,
                                    validate: (value) {
                                      if (value == null || value.isEmpty) {
                                        return "* Required";
                                      } else {
                                        return null;
                                      }
                                    },
                                    label: "Answer",
                                    secured: false,
                                    textCapitalization: TextCapitalization.none,
                                    inputType: TextInputType.text,
                                    controller: answerTwo,
                                    maxLines: 1,
                                    leftPadding: 20.0,
                                  ),
                                ),
                                Container(
                                  padding:
                                      const EdgeInsets.only(left: 30, top: 20),
                                  child: Text(
                                    passwordResetController
                                        .questions.value.value![2].question,
                                    style: TextStyle(
                                      fontSize: 14,
                                      color: Theme.of(context).brightness ==
                                              lightThemeData(context).brightness
                                          ? kContentColorLightTheme
                                          : kContentColorDarkTheme,
                                    ),
                                  ),
                                ),

                                /// TextFromField Confirm password
                                Padding(
                                  padding:
                                      const EdgeInsets.symmetric(vertical: 5.0),
                                  child: TextFromField(
                                    icon: Icons.question_answer_outlined,
                                    enabled: isLoading ? false : true,
                                    validate: (value) {
                                      if (value == null || value.isEmpty) {
                                        return "* Required";
                                      } else {
                                        return null;
                                      }
                                    },
                                    label: "Answer",
                                    secured: false,
                                    textCapitalization: TextCapitalization.none,
                                    inputType: TextInputType.text,
                                    controller: answerThree,
                                    maxLines: 1,
                                    leftPadding: 20.0,
                                  ),
                                ),
                              ],
                            ),
                            const Padding(
                                padding: EdgeInsets.symmetric(vertical: 10.0)),
                            isLoading
                                ? const Padding(
                                    padding: EdgeInsets.only(
                                      left: 160.0,
                                      right: 160,
                                    ),
                                    child: LoadingIndicator(
                                      indicatorType: Indicator.ballBeat,
                                    ),
                                  )
                                : PrimaryButton(
                                    text: 'Save',
                                    onPressed: () async {
                                      if (formKey!.currentState!.validate()) {
                                        if (await passwordResetController
                                            .setSecurityQuestions(
                                          phoneEmail:
                                              widget.user.user.phoneNumber!,
                                          firstAnswer: answerOne.text,
                                          secondAnswer: answerTwo.text,
                                          thirdAnswer: answerThree.text,
                                        )) {
                                          answerOne.clear();
                                          answerTwo.clear();
                                          answerThree.clear();
                                        }
                                      }
                                    },
                                  ),
                          ],
                        ),
                      );
                    },
                  )
                : const EmptyWidget(
                    text: 'You haven\'t set any security questions yet.'),
      ),
    );
  }
}
